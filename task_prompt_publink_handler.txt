PDU* MsgHandler::createPublishLink() {
    QString path = QString::fromUtf8(pdu->caData);
    QString code = QString::number(qrand() % 90000000 + 10000000); // 简单8位随机码
    bool ok = OperateDB::instance().createPublishLink(code, path, "currentUser");
    
    PDU* resp = mkPDU();
    resp->uiType = ENUM_MSG_TYPE_CREATE_PUBLINK_RESPOND;
    memcpy(resp->caData, &ok, sizeof(ok));
    if (ok) memcpy(resp->caMsg, code.toUtf8().constData(), code.size());
    return resp;
}
