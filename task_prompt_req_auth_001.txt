bool OperateDB::handleLogoff(const char* name, const char* pwd)
{
    if(name==nullptr||pwd==nullptr){
        return false;
    }

    PooledSqlConnection connection = DatabaseConnectionPool::instance().acquire();
    if(!connection.isValid()){
        return false;
    }
    QSqlDatabase db = connection.database();
    
    db.transaction();
    QSqlQuery q(db);

    // 1. 验证用户名和密码
    q.prepare("SELECT * FROM user_info WHERE name = ? AND pwd = ?");
    q.addBindValue(QString::fromUtf8(name));
    q.addBindValue(QString::fromUtf8(pwd));
    if(!q.exec() || !q.next()){
        qDebug()<<"logoff: user/pwd mismatch or user not found:"<<QString::fromUtf8(name);
        db.rollback();
        return false;
    }

    // 2. 删除 message_info 中涉及该用户的记录
    q.prepare("DELETE FROM message_info WHERE seed_name = ? OR rev_name = ?");
    q.addBindValue(QString::fromUtf8(name));
    q.addBindValue(QString::fromUtf8(name));
    if(!q.exec()){
        qDebug()<<"logoff: failed to delete messages for:"<<QString::fromUtf8(name);
        db.rollback();
        return false;
    }

    // 3. 删除 friend 表中涉及该用户的所有记录（双向）
    q.prepare("DELETE FROM friend WHERE name = ? OR friendName = ?");
    q.addBindValue(QString::fromUtf8(name));
    q.addBindValue(QString::fromUtf8(name));
    if(!q.exec()){
        qDebug()<<"logoff: failed to delete friend records for:"<<QString::fromUtf8(name);
        db.rollback();
        return false;
    }

    // 4. 删除 user_info 表中的用户记录
    q.prepare("DELETE FROM user_info WHERE name = ? AND pwd = ?");
    q.addBindValue(QString::fromUtf8(name));
    q.addBindValue(QString::fromUtf8(pwd));
    if(!q.exec()){
        qDebug()<<"logoff: failed to delete user_info for:"<<QString::fromUtf8(name);
        db.rollback();
        return false;
    }

    if(!db.commit()){
        qDebug()<<"logoff: commit failed for:"<<QString::fromUtf8(name);
        db.rollback();
        return false;
    }

    qDebug()<<"logoff: successfully deleted user:"<<QString::fromUtf8(name);
    return true;
}
```

PDU *MsgHandler::logoff()
{
    char userName[32]={'\0'};
    char userPwd[32]={'\0'};
    memcpy(userName,pdu->caData,32);
    memcpy(userPwd,pdu->caData+32,32);
    
    qDebug()<<"logoff request for:"<<QString::fromUtf8(userName);
    
    bool ret=OperateDB::instance().handleLogoff(userName,userPwd);
    
    // 如果数据库删除成功，清理用户的文件目录
    if(ret){
        QString userDir=QString("%1/%2").arg(Server::getinstance().filesPath).arg(QString::fromUtf8(userName));
        QDir dir(userDir);
        if(dir.exists()){
            bool removed=dir.removeRecursively();
            qDebug()<<"logoff: remove user dir"<<userDir<<"result:"<<removed;
        }
    }
    
    PDU* respdu=mkPDU();
    respdu->uiType=ENUM_MSG_TYPE_LOGOFF_RESPOND;
    memcpy(respdu->caData,&ret,sizeof(ret));
    return respdu;
}
```

void Client::sendLogoffRequest()
{
    //弹出密码确认框
    bool ok=false;
    QString pwdInput=QInputDialog::getText(this,"注销确认",
        "输入密码以确认注销账户：\n（此操作不可逆，将永久删除您的账户、好友关系和所有文件）",
        QLineEdit::Password,"",&ok);
    if(!ok||pwdInput.isEmpty()){
        return;
    }
    
    //二次确认
    int confirm=QMessageBox::warning(this,"最后确认",
        "确定要永久删除您的账户吗？\n此操作不可撤销！",
        QMessageBox::Yes|QMessageBox::No,QMessageBox::No);
    if(confirm!=QMessageBox::Yes){
        return;
    }
    
    QByteArray passwordHash=passwordHashForProtocol(pwdInput);
    PDU* pdu=mkPDU();
    pdu->uiType=ENUM_MSG_TYPE_LOGOFF_REQUEST;
    copyFixedField(pdu->caData,32,m_LoginName.toUtf8());
    copyExactField(pdu->caData+32,32,passwordHash);
    seedMsg(pdu);
}
